/* Trilho das categorias.
   Celular: o arrasto é feito pelo próprio script (não depende da rolagem nativa,
   que o tema da loja estava travando). Computador: rolagem nativa + arrastar com mouse. */
(function(){
  var rail=document.querySelector('#bfm-vip .bfm-cat .bfm-rail');if(!rail)return;
  var cards=[].slice.call(rail.querySelectorAll('.bfm-cc'));
  cards.concat([].slice.call(rail.querySelectorAll('img'))).forEach(function(el){el.setAttribute('draggable','false')});
  rail.addEventListener('dragstart',function(e){e.preventDefault()});
  var mq=window.matchMedia('(max-width:900px), (pointer:coarse)');
  var anim=0,blockClickUntil=0,drag=null;

  function maxScroll(){return Math.max(0,rail.scrollWidth-rail.clientWidth)}
  function stops(){
    var rr=rail.getBoundingClientRect(),pl=parseFloat(getComputedStyle(rail).paddingLeft)||0,m=maxScroll();
    var list=cards.map(function(c){return Math.min(m,Math.max(0,rail.scrollLeft+c.getBoundingClientRect().left-rr.left-pl))});
    if(list[list.length-1]<m)list.push(m);
    return list;
  }
  function animateTo(x){
    cancelAnimationFrame(anim);
    var from=rail.scrollLeft,d=x-from,t0=0,dur=Math.min(520,220+Math.abs(d)*.6);
    if(Math.abs(d)<1){rail.scrollLeft=x;return}
    function step(now){if(!t0)t0=now;var p=Math.min(1,(now-t0)/dur),e=1-Math.pow(1-p,3);
      rail.scrollLeft=from+d*e;if(p<1)anim=requestAnimationFrame(step);}
    anim=requestAnimationFrame(step);
  }
  function snap(velocity){
    var list=stops(),cur=rail.scrollLeft,i=0;
    list.forEach(function(v,k){if(Math.abs(v-cur)<Math.abs(list[i]-cur))i=k});
    if(Math.abs(velocity)>.3){ /* gesto rápido: vai para o próximo/anterior */
      var dir=velocity<0?1:-1;
      if(dir>0)while(i<list.length-1&&list[i]<=cur+1)i++;else while(i>0&&list[i]>=cur-1)i--;
    }
    animateTo(list[i]);
  }

  function setMode(){rail.classList.toggle('bfm-js',mq.matches)}
  setMode();if(mq.addEventListener)mq.addEventListener('change',setMode);else if(mq.addListener)mq.addListener(setMode);

  /* toque (e caneta) no celular */
  rail.addEventListener('pointerdown',function(e){
    if(e.pointerType==='mouse'&&!mq.matches)return;
    if(e.pointerType!=='mouse'&&!rail.classList.contains('bfm-js'))return;
    if(e.button>0)return;
    cancelAnimationFrame(anim);
    drag={id:e.pointerId,x:e.clientX,y:e.clientY,s:rail.scrollLeft,axis:null,lx:e.clientX,lt:e.timeStamp,v:0,mouse:e.pointerType==='mouse'};
  });
  function move(e){
    if(!drag||e.pointerId!==drag.id)return;
    var dx=e.clientX-drag.x,dy=e.clientY-drag.y;
    if(!drag.axis){
      if(Math.abs(dx)<6&&Math.abs(dy)<6)return;
      drag.axis=Math.abs(dx)>Math.abs(dy)?'x':'y';
      if(drag.axis==='y'){drag=null;return}
      rail.classList.add('bfm-dragging');
    }
    if(e.cancelable)e.preventDefault();
    var dt=e.timeStamp-drag.lt;if(dt>0){drag.v=.8*((e.clientX-drag.lx)/dt)+.2*drag.v;drag.lx=e.clientX;drag.lt=e.timeStamp}
    var m=maxScroll(),x=drag.s-dx;
    if(x<0)x=x/3;else if(x>m)x=m+(x-m)/3; /* resistência nas pontas */
    rail.scrollLeft=x;
  }
  function end(e){
    if(!drag||(e&&e.pointerId!==drag.id))return;
    var wasDrag=drag.axis==='x',v=drag.v;drag=null;rail.classList.remove('bfm-dragging');
    if(wasDrag){blockClickUntil=Date.now()+400;snap(v)}
  }
  rail.addEventListener('pointermove',move,{passive:false});
  rail.addEventListener('pointerup',end);
  rail.addEventListener('pointercancel',end);
  /* impede que scripts do tema briguem com o gesto */
  ['touchstart','touchmove','touchend'].forEach(function(t){rail.addEventListener(t,function(e){e.stopPropagation()},{passive:true})});
  rail.addEventListener('touchmove',function(e){if(drag&&drag.axis==='x'&&e.cancelable)e.preventDefault()},{passive:false});

  /* mouse no computador: arrastar a rolagem nativa */
  var ms=null;
  rail.addEventListener('mousedown',function(e){
    if(mq.matches||e.button!==0||maxScroll()<1)return;
    ms={x:e.clientX,s:rail.scrollLeft,dragged:false};
  });
  window.addEventListener('mousemove',function(e){
    if(!ms)return;if(e.buttons===0){mEnd();return}
    var d=e.clientX-ms.x;if(!ms.dragged&&Math.abs(d)<5)return;
    ms.dragged=true;rail.classList.add('bfm-dragging');e.preventDefault();rail.scrollLeft=ms.s-d;
  },{passive:false});
  function mEnd(){if(ms&&ms.dragged)blockClickUntil=Date.now()+400;ms=null;rail.classList.remove('bfm-dragging')}
  window.addEventListener('mouseup',mEnd);window.addEventListener('blur',mEnd);

  rail.addEventListener('click',function(e){if(Date.now()<blockClickUntil){e.preventDefault();e.stopImmediatePropagation()}},true);
  rail.addEventListener('keydown',function(e){
    if(e.key!=='ArrowLeft'&&e.key!=='ArrowRight')return;e.preventDefault();
    var list=stops(),cur=rail.scrollLeft,i=0;list.forEach(function(v,k){if(Math.abs(v-cur)<Math.abs(list[i]-cur))i=k});
    i=Math.max(0,Math.min(list.length-1,i+(e.key==='ArrowRight'?1:-1)));animateTo(list[i]);
  });
  window.addEventListener('resize',function(){if(rail.classList.contains('bfm-js'))snap(0)});
})();

  function fixedTop(){
    var top=0;
    [].slice.call(document.querySelectorAll('header,#header,.header,[role="banner"],[class*="header"],[id*="header"]')).forEach(function(el){
      if(root.contains(el))return;var cs=getComputedStyle(el);if(cs.position!=='fixed'&&cs.position!=='sticky')return;
      var r=el.getBoundingClientRect();if(r.height>0&&r.height<260&&r.top<=1)top=Math.max(top,r.bottom);
    });
    return top;
  }
  function goToForm(e){

    var top=fixedTop(),vh=window.innerHeight,r=box.getBoundingClientRect();

    if(r.height+32<=room)y=window.scrollY+r.top-top-(room-r.height)/2;

  }
